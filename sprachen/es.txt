arg.woerter = marcas de tiempo por palabra (subtítulos más precisos, JSON con tiempos de palabras)

arg.spur = pista de audio (0 = primera); lista con --spuren
arg.spuren = solo mostrar las pistas de audio, sin transcribir
arg.titel = título del DVD (VTS_xx); predeterminado: el mayor
arg.nur_datei = con VTS_xx_n.VOB, solo este archivo en lugar del título completo
arg.ffmpeg = ruta de ffmpeg.exe (predeterminado: perfil)
arg.threads = núcleos (predeterminado: perfil)
arg.compute = tipo de cálculo: int8, int8_float32, float32 (predeterminado: perfil)
arg.beam = amplitud de búsqueda (predeterminado: perfil; si no, 5)
arg.absatz = nuevo párrafo en el .txt tras esta pausa en s (predeterminado: perfil; si no, 2)
arg.kein_protokoll = no escribir archivo de registro
arg.pruefen = solo comprobar la configuración del perfil
arg.ui = idioma de los mensajes y del registro, p. ej., es o en (predeterminado: como en la interfaz)
arg.sprachen_pruefen = comprobar si faltan entradas en los archivos de idioma
