# ---------------------------------------------------------------- Ligne de commande : aide (--help)
arg.beschreibung = transcription locale (faster-whisper)
arg.audio = entrées audio, vidéo ou DVD (fichier, .iso, dossier VIDEO_TS)
arg.profil = nom ou fichier de profil (.json)
arg.modell = nom court, nom ou dossier du modèle (par défaut : profil)
arg.sprache = langue de l'enregistrement : de, en, … ou auto (par défaut : profil)
arg.begriffe = termes, séparés par des virgules
arg.begriffe_datei = fichier texte de termes (par défaut : profil)
arg.kontext = phrase sur le thème comme indice de reconnaissance (par défaut : profil)
arg.ausgabe = dossier de sortie (par défaut : profil, sinon à côté du fichier audio)
arg.test = seulement les n premières secondes (essai)
arg.woerter = horodatage par mot (sous-titres plus précis, JSON avec temps des mots)
arg.raster = grille temporelle en secondes, p. ex. 0.5 1 5 (active --woerter)
arg.spur = piste audio (0 = première) ; liste avec --spuren
arg.spuren = afficher seulement les pistes audio, sans transcrire
arg.titel = titre du DVD (VTS_xx) ; par défaut : le plus grand
arg.nur_datei = pour VTS_xx_n.VOB, seulement ce fichier au lieu du titre entier
arg.ffmpeg = chemin de ffmpeg.exe (par défaut : profil)
arg.threads = cœurs (par défaut : profil)
arg.compute = type de calcul : int8, int8_float32, float32 (par défaut : profil)
arg.beam = largeur de recherche (par défaut : profil, sinon 5)
arg.absatz = nouveau paragraphe dans le .txt après cette pause en s (par défaut : profil, sinon 2)
arg.kein_protokoll = ne pas écrire de journal
arg.pruefen = vérifier seulement la configuration du profil
arg.ui = langue des messages et du journal, p. ex. fr ou en (par défaut : comme dans l'interface)
arg.sprachen_pruefen = vérifier les entrées manquantes des fichiers de langue
